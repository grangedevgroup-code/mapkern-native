cmake_minimum_required(VERSION 3.24)

project(mapkern VERSION 0.1.0 LANGUAGES C CXX)

if(APPLE)
  enable_language(OBJC OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(MAPKERN_RENDER "Build the renderer" ON)
option(MAPKERN_VIEWER "Build the desktop viewer" ON)
option(MAPKERN_TESTS "Build the tests" ON)
option(MAPKERN_SHARED "Build the C API as a shared library" ON)
option(MAPKERN_GLES "Use OpenGL ES 3 on Linux" OFF)
set(MAPKERN_GFX "" CACHE STRING "Graphics backend: D3D11, METAL, GLCORE or GLES3 (empty picks the platform default)")

if(MAPKERN_GFX STREQUAL "")
  if(WIN32)
    set(MAPKERN_GFX D3D11)
  elseif(APPLE)
    set(MAPKERN_GFX METAL)
  elseif(ANDROID OR MAPKERN_GLES)
    set(MAPKERN_GFX GLES3)
  else()
    set(MAPKERN_GFX GLCORE)
  endif()
endif()

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /Zc:__cplusplus /wd4100 /wd4127 /wd4244 /wd4267 /wd4305 /wd4996)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
endif()

include(cmake/Sources.cmake)

add_library(mapkern_core STATIC ${MAPKERN_CORE_SOURCES})
target_include_directories(mapkern_core PUBLIC include src PRIVATE third_party)
find_package(Threads REQUIRED)
target_link_libraries(mapkern_core PUBLIC Threads::Threads)
include(cmake/Platform.cmake)
include(cmake/Dependencies.cmake)

if(MAPKERN_RENDER)
  add_library(mapkern STATIC ${MAPKERN_RENDER_SOURCES})
  target_link_libraries(mapkern PUBLIC mapkern_core)
  target_include_directories(mapkern PUBLIC third_party)
  target_compile_definitions(mapkern PUBLIC SOKOL_${MAPKERN_GFX})
  include(cmake/Graphics.cmake)
  if(MAPKERN_SHARED AND EXISTS ${PROJECT_SOURCE_DIR}/src/capi/mapkern_c.cpp)
    add_library(mapkern_c SHARED src/capi/mapkern_c.cpp src/capi/hwnd_view.cpp)
    target_link_libraries(mapkern_c PRIVATE mapkern)
    if(WIN32)
      target_link_libraries(mapkern_c PRIVATE d3d11 dxgi)
    endif()
    target_compile_definitions(mapkern_c PRIVATE MAPKERN_BUILDING)
    set_target_properties(mapkern_c PROPERTIES OUTPUT_NAME mapkern ARCHIVE_OUTPUT_NAME mapkern_c)
  endif()
  if(ANDROID)
    add_library(mapkern-jni SHARED src/platform/android/jni.cpp)
    target_link_libraries(mapkern-jni PRIVATE mapkern)
    target_link_options(mapkern-jni PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
  if(MAPKERN_VIEWER AND NOT ANDROID AND NOT IOS)
    add_subdirectory(apps/viewer)
  endif()
  if(WIN32)
    add_subdirectory(apps/render)
  endif()
endif()

if(MAPKERN_TESTS)
  enable_testing()
  add_subdirectory(test)
endif()
